6/** Clone a hosted repository into a task's workspace. */
7export async function cloneRepository(project: string, into: string, signal: AbortSignal) {
8 await mkdir(into, { recursive: true });
9 const child = spawn('git', ['clone', '--quiet', repositoryPath(project), into], {
10 stdio: ['ignore', 'ignore', 'pipe'],
11 });
12 const onAbort = () => child.kill('SIGTERM');
13 signal.addEventListener('abort', onAbort, { once: true });
14
15 try {
16 const code = await exited(child);
17 if (signal.aborted) throw new CloneCancelled(project);
18 if (code !== 0) throw new Error(`git clone exited with ${code}`);
19 } catch (error) {
20 // A cancelled clone leaves a partial checkout behind.
21 await rm(join(into, '.git'), { recursive: true, force: true });
22 throw error;
23 } finally {
24 signal.removeEventListener('abort', onAbort);
25 }
26}

28/** Wait for the abort to settle before callers assert on it. */
29export async function waitForAbort(child: ReturnType<typeof spawn>, timeoutMs = 5000) {
30 if (child.exitCode !== null) return child.exitCode;
31 return Promise.race([exited(child), delay(timeoutMs).then(() => null)]);
32}
